-- 1. Total stock, estimated acquisition and maintenance cost
SELECT SUM(quantity) AS total_units, SUM(quantity * unit_cost) AS acquisition_value,
       SUM(maintenance_cost) AS maintenance_cost FROM equipment;

-- 2. Quantity by department
SELECT department, SUM(quantity) AS units FROM equipment GROUP BY department ORDER BY units DESC;

-- 3. Quantity by equipment type
SELECT equipment_type, SUM(quantity) AS units FROM equipment GROUP BY equipment_type ORDER BY units DESC;

-- 4. Repair priority (record count and affected quantity)
SELECT COUNT(*) AS records_to_repair, SUM(quantity) AS units_to_repair
FROM equipment WHERE condition = 'Needs Repair';

-- 5. Estimated acquisition value per category
SELECT equipment_type, SUM(quantity * unit_cost) AS acquisition_value
FROM equipment GROUP BY equipment_type ORDER BY acquisition_value DESC;

-- 6. Highest single maintenance expense
SELECT asset_id, equipment_type, department, maintenance_cost
FROM equipment ORDER BY maintenance_cost DESC LIMIT 1;
